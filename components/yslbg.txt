        blossoms: ['#ffffff', '#fdf4ff', '#fffbeb'],

        monolithEdge: 'rgba(165, 180, 252, 0.45)'
      },
      steelMonolith: {
        glow: 'rgba(148, 163, 184, 0.15)',

        shards: ['rgba(100, 116, 139, 0.4)', 'rgba(148, 163, 184, 0.6)', 'rgba(226, 232, 240, 0.7)'],
        blossoms: ['#f8fafc', '#e2e8f0', '#cbd5e1'],

      }

    const mouse = {
      x: window.innerWidth * 0.5,

    // ================= ARCHITECTURAL MONOLITHS (YSL TOTEM GEOMETRY) =================
    let monoliths = [];

    class Monolith {
      constructor(x, y, w, h, depthFactor, bevel) {
        this.x = x;
        this.y = y;
        this.w = w;
        this.h = h;
        this.depthFactor = depthFactor; // Parallax response
        this.bevel = bevel;
      }

      draw(ctx, parallaxX, parallaxY, pal) {
        ctx.save();
        const shiftX = (parallaxX - width / 2) * this.depthFactor;
        const shiftY = (parallaxY - height / 2) * this.depthFactor * 0.4;

        ctx.translate(this.x + shiftX, this.y + shiftY);

        const presence = config.monolithFactor;
        const grad = ctx.createLinearGradient(0, 0, this.w, this.h);
        grad.addColorStop(0, `rgba(16, 24, 38, ${0.45 * presence})`);
        grad.addColorStop(0.5, `rgba(6, 10, 16, ${0.75 * presence})`);
        grad.addColorStop(1, `rgba(2, 4, 6, ${0.9 * presence})`);

        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.rect(0, 0, this.w, this.h);
        ctx.fill();

        // Sleek specular edge light
        ctx.strokeStyle = pal.monolithEdge;
        ctx.lineWidth = 1;
        ctx.globalAlpha = 0.5 * presence;
        ctx.strokeRect(0, 0, this.w, this.h);

        // Center vertical refraction facet slit (reminiscent of the YSL MYSLF black monolith center facet)
        const centerGrad = ctx.createLinearGradient(this.w * 0.48, 0, this.w * 0.52, 0);
        centerGrad.addColorStop(0, 'rgba(255, 255, 255, 0)');
        centerGrad.addColorStop(0.5, `rgba(255, 255, 255, ${0.25 * presence})`);
        centerGrad.addColorStop(1, 'rgba(255, 255, 255, 0)');
        ctx.fillStyle = centerGrad;
        ctx.fillRect(this.w * 0.48, 0, this.w * 0.04, this.h);

        ctx.restore();
      }
    }

    function createMonoliths() {
      monoliths = [];
      const count = Math.max(4, Math.floor(width / 320));
      const spacing = width / count;

      for (let i = 0; i < count; i++) {
        const w = 90 + Math.random() * 80;
        const h = height * (0.65 + Math.random() * 0.3);
        const x = spacing * i + (spacing - w) / 2 + (Math.random() - 0.5) * 40;
        const y = height - h + (Math.random() - 0.5) * 40;
        const depth = 0.015 + (i % 2) * 0.015;
        monoliths.push(new Monolith(x, y, w, h + 50, depth, 4));
      }
    }

    // ================= MAIN PARTICLES: DROPLETS, GLASS FACETS & BLOSSOMS =================
    class Particle {
      constructor(isInitial = false) {
        this.reset(isInitial);
      }

      reset(isInitial = false) {
        const r = Math.random();
        if (r > 0.55) {
          this.type = 0; // Pure Ozone Condensation Droplet
        } else if (r > 0.25) {
          this.type = 1; // 3D Crystalline Glass Shard / Hexagonal Facet
        } else {
          this.type = 2; // Delicate White Orange Blossom Petal
        }

        this.x = isInitial ? Math.random() * width : Math.random() * width * 1.3 - width * 0.15;
        this.y = isInitial ? Math.random() * height : -30 - Math.random() * 60;

        this.depth = 0.4 + Math.random() * 0.85;
        this.angle = Math.random() * Math.PI * 2;
        this.angularSpeed = (Math.random() - 0.5) * 0.035;

        if (this.type === 0) {
          // Water Droplet
          this.size = (2.2 + Math.random() * 4.5) * this.depth;
          this.vx = (Math.random() - 0.5) * 0.5;
          this.vy = (1.2 + Math.random() * 2.2) * this.depth;
          this.life = 1.0;
          this.decay = 0.0006 + Math.random() * 0.0014;
        } else if (this.type === 1) {
          // Glass Facet / Crystalline Shard
          this.size = (7 + Math.random() * 12) * this.depth;
          this.aspect = 0.4 + Math.random() * 0.5;
          this.vx = 0.3 + Math.random() * 0.9;
          this.vy = (0.8 + Math.random() * 1.6) * this.depth;
          this.flip = Math.random() * Math.PI;
          this.flipSpeed = 0.02 + Math.random() * 0.035;
          this.life = 1.0;
          this.decay = 0.0008 + Math.random() * 0.0016;
        } else {
          // Orange Blossom Petal
          this.size = (9 + Math.random() * 13) * this.depth;
          this.vx = 0.4 + Math.random() * 1.0;
          this.vy = (0.7 + Math.random() * 1.4) * this.depth;
          this.flip = Math.random() * Math.PI;
          this.flipSpeed = 0.015 + Math.random() * 0.03;
          this.sway = Math.random() * Math.PI * 2;
          this.swaySpeed = 0.02 + Math.random() * 0.03;
          this.life = 1.0;
          this.decay = 0.0007 + Math.random() * 0.0015;
        }

        this.updateColor();
      }

      updateColor() {
        const pal = palettes[config.activePalette];
        if (this.type === 0) {
          this.color = pal.droplets[Math.floor(Math.random() * pal.droplets.length)];
        } else if (this.type === 1) {
          this.color = pal.shards[Math.floor(Math.random() * pal.shards.length)];
        } else {
          this.color = pal.blossoms[Math.floor(Math.random() * pal.blossoms.length)];
        }
      }

      update() {
        this.life -= this.decay;
        this.angle += this.angularSpeed * config.windFactor;

        if (this.type === 0) {
          this.x += this.vx * config.windFactor;
          this.y += this.vy * config.windFactor;
        } else if (this.type === 1) {
          this.flip += this.flipSpeed * config.windFactor;
          this.x += this.vx * config.windFactor;
          this.y += this.vy * config.windFactor;
        } else {
          this.sway += this.swaySpeed * config.windFactor;
          this.flip += this.flipSpeed * config.windFactor;
          this.x += (this.vx + Math.sin(this.sway) * 1.2) * config.windFactor;
          this.y += this.vy * config.windFactor;
        }

        // Pointer Magnetic Wake / Fluid Air Displacement
        const dx = this.x - mouse.x;
        const dy = this.y - mouse.y;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < config.pullRadius) {
          const norm = 1 - dist / config.pullRadius;
          const angle = Math.atan2(dy, dx);

          if (config.isVortexActive) {
            const tangent = angle + Math.PI / 2;
            const swirl = norm * 4.6 * this.depth;
            const pull = norm * 1.5 * this.depth;
            this.x += Math.cos(tangent) * swirl - Math.cos(angle) * pull;
            this.y += Math.sin(tangent) * swirl - Math.sin(angle) * pull;
          } else {
            const push = norm * 5.8 * this.depth;
            this.x += Math.cos(angle) * push;
            this.y += Math.sin(angle) * push;
          }
        }

        // Reset off-screen or end of life
        if (this.life <= 0 || this.y > height + 50 || this.x > width + 60 || this.x < -60) {
          this.reset(false);
        }
      }

      draw(c) {
        c.save();
        c.translate(this.x, this.y);
        c.rotate(this.angle);

        if (this.type === 0) {
          // Ozone Water Droplet / Crisp Condensation Sphere
          c.globalAlpha = Math.max(0.1, this.life * 0.85);
          c.shadowColor = '#38bdf8';
          c.shadowBlur = 9 * config.radianceFactor;

          // Liquid Body
          c.fillStyle = this.color;
          c.beginPath();
          c.arc(0, 0, this.size, 0, Math.PI * 2);
          c.fill();

          // Specular Glint Highlight
          c.fillStyle = '#ffffff';
          c.beginPath();
          c.arc(-this.size * 0.35, -this.size * 0.35, this.size * 0.35, 0, Math.PI * 2);
          c.fill();

        } else if (this.type === 1) {
          // Geometric Glass Facet / Ice Shard
          const flipScale = Math.sin(this.flip);
          c.scale(1, Math.abs(flipScale) < 0.15 ? 0.15 : flipScale);
          c.globalAlpha = Math.max(0.1, this.life * 0.85);

          const s = this.size;
          c.shadowColor = '#7dd3fc';
          c.shadowBlur = 10 * config.radianceFactor;

          // Rhombus / Facet Shape
          c.fillStyle = this.color;
          c.beginPath();
          c.moveTo(0, -s);
          c.lineTo(s * this.aspect, 0);
          c.lineTo(0, s);
          c.lineTo(-s * this.aspect, 0);
          c.closePath();
          c.fill();

          // Precision Edge Wire
          c.strokeStyle = '#ffffff';
          c.lineWidth = 0.8;
          c.globalAlpha = 0.55;
          c.stroke();

        } else {
          // Orange Blossom White Petal
          const flipScale = Math.cos(this.flip);
          c.scale(Math.abs(flipScale) < 0.15 ? 0.15 : flipScale, 1);
          c.globalAlpha = Math.max(0.1, this.life * 0.82);

          const s = this.size;
          c.shadowColor = '#e0f2fe';
          c.shadowBlur = 12 * config.radianceFactor;

          // Curved floral petal contour
          c.fillStyle = this.color;
          c.beginPath();
          c.moveTo(0, -s);
          c.bezierCurveTo(s * 0.5, -s * 0.6, s * 0.6, s * 0.2, 0, s);
          c.bezierCurveTo(-s * 0.6, s * 0.2, -s * 0.5, -s * 0.6, 0, -s);
          c.fill();

          // Tender central spine
          c.strokeStyle = '#38bdf8';
          c.lineWidth = 0.7;
          c.globalAlpha = 0.35;
          c.beginPath();
          c.moveTo(0, -s * 0.8);
          c.lineTo(0, s * 0.8);
          c.stroke();
        }

        c.restore();
      }
    }

    // ================= COOL ELECTRIC SILVER / CYAN TRAIL =================
    const sparks = [];
    function addSparks(x, y, count = 1) {
      const pal = palettes[config.activePalette];
      for (let i = 0; i < count; i++) {
        if (sparks.length > 80) sparks.shift();
        sparks.push({
          x: x + (Math.random() - 0.5) * 14,
          y: y + (Math.random() - 0.5) * 14,
          vx: (Math.random() - 0.5) * 1.8,
          vy: (Math.random() - 0.5) * 1.8 - 0.6,
          radius: 1.2 + Math.random() * 2.2,
          color: pal.sparks[Math.floor(Math.random() * pal.sparks.length)],
          life: 1.0,
          decay: 0.025 + Math.random() * 0.03
        });
      }
    }

    let particles = [];

    function syncParticleCount() {
      if (particles.length < config.particleCount) {
        while (particles.length < config.particleCount) {
          particles.push(new Particle(false));
        }
      } else if (particles.length > config.particleCount) {
        particles.splice(config.particleCount);
      }
    }

    function resize() {
      width = archCanvas.width = mainCanvas.width = sparkCanvas.width = window.innerWidth;
      height = archCanvas.height = mainCanvas.height = sparkCanvas.height = window.innerHeight;

      createMonoliths();
      particles = Array.from({ length: config.particleCount }, () => new Particle(true));
    }

    // ================= MAIN ANIMATION LOOP =================
    function render() {
      mouse.x += (mouse.targetX - mouse.x) * 0.08;
      mouse.y += (mouse.targetY - mouse.y) * 0.08;

      // Update Cursor Cool Bloom
      if (ambientGlow) {
        const xPercent = (mouse.x / width) * 100;
        const yPercent = (mouse.y / height) * 100;
        ambientGlow.style.setProperty('--mouse-x', `${xPercent}%`);
        ambientGlow.style.setProperty('--mouse-y', `${yPercent}%`);
      }

      // 1. Render Architectural Monoliths with Parallax
      archCtx.clearRect(0, 0, width, height);
      const activePal = palettes[config.activePalette];
      for (let mono of monoliths) {
        mono.draw(archCtx, mouse.x, mouse.y, activePal);
      }

      // 2. Render Droplets, Shards, and Blossom Petals
      mainCtx.clearRect(0, 0, width, height);
      for (let i = 0; i < particles.length; i++) {
        particles[i].update();
        particles[i].draw(mainCtx);
      }

      // 3. Render Cool Spark Trail
      sparkCtx.clearRect(0, 0, width, height);
      for (let i = sparks.length - 1; i >= 0; i--) {
        const sp = sparks[i];
        sp.x += sp.vx;
        sp.y += sp.vy;
        sp.life -= sp.decay;

        if (sp.life <= 0) {
          sparks.splice(i, 1);
          continue;
        }

        sparkCtx.save();
        sparkCtx.globalAlpha = sp.life * 0.9;
        sparkCtx.fillStyle = sp.color;
        sparkCtx.shadowColor = sp.color;
        sparkCtx.shadowBlur = 12 * config.radianceFactor;
        sparkCtx.beginPath();
        sparkCtx.arc(sp.x, sp.y, sp.radius * sp.life, 0, Math.PI * 2);
        sparkCtx.fill();
        sparkCtx.restore();
      }

      requestAnimationFrame(render);
    }

    // ================= POINTER & COLD SHOCKWAVE =================
    function handlePointer(e) {
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;

      mouse.targetX = clientX;
      mouse.targetY = clientY;

      addSparks(clientX, clientY, 1);
      if (mouse.isDown) addSparks(clientX, clientY, 2);
    }

    window.addEventListener('mousemove', handlePointer, { passive: true });
    window.addEventListener('touchmove', handlePointer, { passive: true });

    window.addEventListener('mousedown', (e) => {
      if (e.target.closest('#ambientPopover, #toggleControlsBtn')) return;
      mouse.isDown = true;
      triggerColdBlast(e.clientX, e.clientY, 20);
    });

    window.addEventListener('mouseup', () => { mouse.isDown = false; });

    // Cold Blast Shockwave
    function triggerColdBlast(x, y, count = 22) {
      addSparks(x, y, 28);
      for (let i = 0; i < count; i++) {
        const p = new Particle(false);
        p.x = x + (Math.random() - 0.5) * 35;
        p.y = y + (Math.random() - 0.5) * 35;
        const angle = Math.random() * Math.PI * 2;
        const speed = 3.2 + Math.random() * 7.5;
        p.vx = Math.cos(angle) * speed;
        p.vy = Math.sin(angle) * speed;
        p.life = 1.0;
        p.updateColor();
        particles.push(p);
      }

      if (particles.length > config.particleCount + 35) {
        particles.splice(0, count);
      }
    }

    // ================= UI CONSOLE LOGIC =================

    const particleDensityInput = document.getElementById('particleDensity');
    const coolRadianceInput = document.getElementById('coolRadiance');
    const monolithPresenceInput = document.getElementById('monolithPresence');

    const triggerBurstBtn = document.getElementById('triggerBurstBtn');
    const paletteButtons = document.querySelectorAll('.palette-btn');

    function toggleMenu(show) {
      if (show) {
        popover.classList.remove('hidden');
        requestAnimationFrame(() => {
          popover.classList.remove('scale-95', 'opacity-0');
          popover.classList.add('scale-100', 'opacity-100');
        });
      } else {
        popover.classList.remove('scale-100', 'opacity-100');
        popover.classList.add('scale-95', 'opacity-0');
        setTimeout(() => popover.classList.add('hidden'), 250);
      }
    }

    toggleBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const isClosed = popover.classList.contains('hidden');
      toggleMenu(isClosed);
    });

      document.getElementById('windVal').textContent = `${config.windFactor.toFixed(1)}x Crisp`;

      document.getElementById('densityVal').textContent = `${config.particleCount > 110 ? 'Dense Ozone' : config.particleCount > 65 ? 'Balanced' : 'Minimal'} (${config.particleCount})`;
      syncParticleCount();
    });

    coolRadianceInput.addEventListener('input', (e) => {
      config.radianceFactor = parseFloat(e.target.value);
      document.getElementById('radianceVal').textContent =
        config.radianceFactor > 1.8 ? 'Hyper Neon' : config.radianceFactor > 1.0 ? 'Electric Cool' : 'Subtle Steel';
    });

      config.monolithFactor = parseFloat(e.target.value);
      document.getElementById('monolithVal').textContent =
        config.monolithFactor > 0.8 ? 'High Contrast' : config.monolithFactor > 0.45 ? 'Glossy Specular' : 'Misty Silhouette';
    });

    // Palette Switcher
    paletteButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        paletteButtons.forEach(b => {
          b.classList.remove('active', 'border-cyan-400/80', 'bg-cyan-950/40', 'text-cyan-200', 'shadow-[0_0_12px_rgba(56,189,248,0.25)]');
          b.classList.add('border-slate-700', 'bg-slate-900/60', 'text-slate-400');
        });
        btn.classList.add('active', 'border-cyan-400/80', 'bg-cyan-950/40', 'text-cyan-200', 'shadow-[0_0_12px_rgba(56,189,248,0.25)]');
        btn.classList.remove('border-slate-700', 'bg-slate-900/60', 'text-slate-400');

        config.activePalette = btn.getAttribute('data-palette');
        particles.forEach(p => p.updateColor());
      });
    });

    // Wind Dynamics Toggle
    toggleVortexBtn.addEventListener('click', () => {
      config.isVortexActive = !config.isVortexActive;
      toggleVortexBtn.textContent = config.isVortexActive ? 'Active Wake' : 'Free Gust';
      toggleVortexBtn.classList.toggle('border-cyan-400/40');
      toggleVortexBtn.classList.toggle('border-slate-500/40');
    });

    // Cold Blast Burst
    triggerBurstBtn.addEventListener('click', () => {
      triggerColdBlast(width * 0.5, height * 0.45, 32);
      particles.forEach(p => {
        p.vx = (Math.random() - 0.5) * 7.0;
        p.vy = (Math.random() - 0.5) * 6.0 + 2.0;
      });
    });

    // Close popover on outside click
    document.addEventListener('click', (e) => {
      if (!popover.contains(e.target) && !toggleBtn.contains(e.target)) {
        toggleMenu(false);
      }
    });

    window.addEventListener('resize', resize);

    window.onload = function() {
      resize();
      render();
    };
