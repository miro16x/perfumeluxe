    // Mouse Tracking

    // ================= JASMINE BLOSSOM / PETAL CLASS =================
    class JasmineElement {
      constructor(isInitial = false) {
        this.reset(isInitial);
      }

      reset(isInitial = false) {
        this.x = isInitial ? Math.random() * width : Math.random() * width * 1.3 - width * 0.15;
        this.y = isInitial ? Math.random() * height : -80 - Math.random() * 80;

        // Is it a full 5-star nocturnal jasmine blossom or a single undulating petal?
        this.isFullBlossom = Math.random() > 0.45;

        // Depth scale (0.45 far back & soft, 1.35 foreground & bold)
        this.depth = 0.45 + Math.random() * 0.9;
        this.baseSize = (this.isFullBlossom ? 14 + Math.random() * 12 : 9 + Math.random() * 10) * this.depth;
        this.size = this.baseSize;

        // 3D rotation and tumbling
        this.angle = Math.random() * Math.PI * 2;
        this.angularSpeed = (Math.random() - 0.5) * 0.024;
        this.flip = Math.random() * Math.PI;
        this.flipSpeed = 0.012 + Math.random() * 0.024;

        this.vx = (0.2 + Math.random() * 0.6) * this.depth;
        this.vy = (0.35 + Math.random() * 0.8) * this.depth;

        this.opacity = (0.55 + Math.random() * 0.4) * (this.depth > 1 ? 0.95 : 0.8);
        this.swayPhase = Math.random() * Math.PI * 2;
        this.swaySpeed = 0.015 + Math.random() * 0.02;

        this.updateColor();
      }

      updateColor() {
        const pal = palettes[config.activePalette];
        this.colorSet = pal.petals[Math.floor(Math.random() * pal.petals.length)];
        this.coreColor = pal.core;
      }

      update() {
        this.swayPhase += this.swaySpeed * config.speedFactor;
        this.angle += this.angularSpeed * config.speedFactor;
        this.flip += this.flipSpeed * config.speedFactor;

        const sway = Math.sin(this.swayPhase) * (1.2 * this.depth);
        this.x += (this.vx * config.speedFactor + config.windX + sway);
        this.y += (this.vy * config.speedFactor + config.windY);

        // Interactive Distance to Cursor
        const dx = this.x - mouse.x;
        const dy = this.y - mouse.y;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < config.mouseRepelRadius) {
          const normDist = (1 - dist / config.mouseRepelRadius);
          
          if (config.isVortexActive) {
            // Scent vortex swirl around cursor
            const angle = Math.atan2(dy, dx);
            const tangentAngle = angle + Math.PI / 2;
            const push = normDist * 3.8 * this.depth;
            const swirl = normDist * 2.8 * this.depth * config.swirlForce;

            this.x += Math.cos(angle) * push + Math.cos(tangentAngle) * swirl;
            this.y += Math.sin(angle) * push + Math.sin(tangentAngle) * swirl;
            this.angle += 0.05 * normDist;
          } else {
            // Pure fluid repulsion
            const force = normDist * 4.5 * this.depth;
            const angle = Math.atan2(dy, dx);
            this.x += Math.cos(angle) * force;
            this.y += Math.sin(angle) * force;
          }
        }

        // Screen wrap
        if (this.y > height + 90 || this.x > width + 130 || this.x < -140) {
          this.reset(false);
        }
      }

      draw(c) {
        c.save();
        c.translate(this.x, this.y);
        c.rotate(this.angle);

        const verticalFlip = Math.sin(this.flip);
        c.scale(1, Math.abs(verticalFlip) < 0.12 ? 0.12 : verticalFlip);

        c.globalAlpha = this.opacity;

        if (this.isFullBlossom) {
          // ================= STAR JASMINE BLOSSOM (5 PETALS) =================
          const numPetals = 5;
          const petalLength = this.size;
          const petalWidth = this.size * 0.42;

          // Warm Pistil Back-Glow
          c.shadowColor = palettes[config.activePalette].glow;
          c.shadowBlur = 10 * this.depth;

          for (let p = 0; p < numPetals; p++) {
            c.save();
            c.rotate((p * Math.PI * 2) / numPetals);

            // Elliptic Star-Shaped Jasmine Petal with subtle propeller tilt
            c.beginPath();
            c.moveTo(0, 0);
            c.bezierCurveTo(petalWidth, -petalLength * 0.4, petalWidth * 0.8, -petalLength * 0.85, 0, -petalLength);
            c.bezierCurveTo(-petalWidth * 0.4, -petalLength * 0.85, -petalWidth * 0.5, -petalLength * 0.4, 0, 0);
            c.closePath();

            // Warm Ivory to Golden Vein Gradient
            const grad = c.createLinearGradient(0, 0, 0, -petalLength);
            grad.addColorStop(0, this.coreColor);
            grad.addColorStop(0.35, this.colorSet.fill);
            grad.addColorStop(1, this.colorSet.stroke);

            c.fillStyle = grad;
            c.fill();

            // Delicate Central Jasmine Spine
            c.beginPath();
            c.moveTo(0, 0);
            c.quadraticCurveTo(petalWidth * 0.1, -petalLength * 0.5, 0, -petalLength * 0.85);
            c.strokeStyle = this.colorSet.stroke;
            c.lineWidth = 0.5;
            c.stroke();

            c.restore();
          }

          // Central Golden Nectar Pistil
          c.beginPath();
          c.arc(0, 0, this.size * 0.16, 0, Math.PI * 2);
          c.fillStyle = this.coreColor;
          c.shadowColor = this.coreColor;
          c.shadowBlur = 8;
          c.fill();

        } else {
          // ================= SINGLE FALLING JASMINE PETAL =================
          c.beginPath();
          c.moveTo(0, -this.size * 0.9);
          c.bezierCurveTo(this.size * 0.7, -this.size * 0.7, this.size * 0.85, this.size * 0.55, 0, this.size * 0.95);
          c.bezierCurveTo(-this.size * 0.85, this.size * 0.55, -this.size * 0.7, -this.size * 0.7, 0, -this.size * 0.9);
          c.closePath();

          const grad = c.createLinearGradient(-this.size * 0.3, -this.size * 0.8, this.size * 0.3, this.size * 0.8);
          grad.addColorStop(0, '#ffffff');
          grad.addColorStop(0.4, this.colorSet.fill);
          grad.addColorStop(1, this.colorSet.stroke);

          c.fillStyle = grad;
          c.shadowColor = this.colorSet.shadow;
          c.shadowBlur = 8 * this.depth;
          c.fill();

          // Delicate Vein
          c.beginPath();
          c.moveTo(0, -this.size * 0.6);
          c.quadraticCurveTo(this.size * 0.1, 0, 0, this.size * 0.75);
          c.strokeStyle = this.colorSet.stroke;
          c.lineWidth = 0.55;
          c.stroke();
        }

        c.restore();
      }
    }

    // ================= SUSPENDED WARM AMBER EMBERS & HONEY ORBS =================
    class AmberEmber {
      constructor() {
        this.reset();
      }

      reset() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.size = 0.9 + Math.random() * 2.8;
        this.alpha = 0.2 + Math.random() * 0.8;
        this.pulse = Math.random() * Math.PI * 2;
        this.pulseSpeed = 0.02 + Math.random() * 0.045;
        this.vx = (Math.random() - 0.5) * 0.35;
        this.vy = -0.25 - Math.random() * 0.5; // Natural upward thermal drift
        
        const pal = palettes[config.activePalette];
        this.color = pal.emberColors[Math.floor(Math.random() * pal.emberColors.length)];
      }

      update() {
        this.pulse += this.pulseSpeed * config.shimmerFactor;
        this.y += this.vy * config.shimmerFactor;
        this.x += this.vx * config.shimmerFactor;

        const dx = this.x - mouse.x;
        const dy = this.y - mouse.y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 100) {
          const angle = Math.atan2(dy, dx);
          this.x += Math.cos(angle) * 1.8;
          this.y += Math.sin(angle) * 1.8;
        }

        if (this.y < -20 || this.x < -10 || this.x > width + 10) {
          this.reset();
          this.y = height + 10;
        }
      }

      draw(c) {
        const curAlpha = Math.max(0, (Math.sin(this.pulse) * 0.5 + 0.5) * this.alpha);
        c.save();
        c.globalAlpha = curAlpha;
        c.fillStyle = this.color;
        c.shadowColor = this.color;
        c.shadowBlur = 8;
        c.beginPath();
        c.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        c.fill();
        c.restore();
      }
    }

    // ================= INTERACTIVE CURSOR SPARKLE TRAIL =================
    const stardustTrail = [];
    function addStardust(x, y, count = 1) {
      const pal = palettes[config.activePalette];
      for (let i = 0; i < count; i++) {
        if (stardustTrail.length > 60) stardustTrail.shift();
        stardustTrail.push({
          x: x + (Math.random() - 0.5) * 22,
          y: y + (Math.random() - 0.5) * 22,
          radius: 1.2 + Math.random() * 2.8,
          color: pal.emberColors[Math.floor(Math.random() * pal.emberColors.length)],
          life: 1.0,
          decay: 0.025 + Math.random() * 0.025
        });
      }
    }

    let blossoms = [];
    let embers = [];

    function syncBlossomCount() {
      if (blossoms.length < config.blossomCount) {
        while (blossoms.length < config.blossomCount) {
          blossoms.push(new JasmineElement(false));
        }
      } else if (blossoms.length > config.blossomCount) {
        blossoms.splice(config.blossomCount);
      }
    }

    function resize() {
      width = canvas.width = emberCanvas.width = window.innerWidth;
      height = canvas.height = emberCanvas.height = window.innerHeight;

      blossoms = Array.from({ length: config.blossomCount }, () => new JasmineElement(true));
      embers = Array.from({ length: config.emberCount }, () => new AmberEmber());
    }

    // ================= ANIMATION TICK =================
    function render() {
      // Smooth cursor lerp
      mouse.vx = mouse.targetX - mouse.x;
      mouse.vy = mouse.targetY - mouse.y;
      mouse.x += mouse.vx * 0.1;
      mouse.y += mouse.vy * 0.1;

      // Update ambient candlelight bloom CSS position
      if (ambientGlow) {
        const xPercent = (mouse.x / width) * 100;
        const yPercent = (mouse.y / height) * 100;
        ambientGlow.style.setProperty('--mouse-x', `${xPercent}%`);
        ambientGlow.style.setProperty('--mouse-y', `${yPercent}%`);
      }

      ctx.clearRect(0, 0, width, height);

      // Render Jasmine Blossoms & Petals
      for (let i = 0; i < blossoms.length; i++) {
        blossoms[i].update();
        blossoms[i].draw(ctx);
      }

      // Render Warm Amber Embers
      for (let i = 0; i < embers.length; i++) {
        embers[i].update();
        embers[i].draw(ctx);
      }

      // Render Stardust Trail
      emberCtx.clearRect(0, 0, width, height);
      for (let i = stardustTrail.length - 1; i >= 0; i--) {
        const pt = stardustTrail[i];
        pt.life -= pt.decay;
        if (pt.life <= 0) {
          stardustTrail.splice(i, 1);
          continue;
        }

        emberCtx.save();
        emberCtx.globalAlpha = pt.life * 0.9;
        emberCtx.fillStyle = pt.color;
        emberCtx.shadowColor = pt.color;
        emberCtx.shadowBlur = 10;
        emberCtx.beginPath();
        emberCtx.arc(pt.x, pt.y, pt.radius * pt.life, 0, Math.PI * 2);
        emberCtx.fill();
        emberCtx.restore();
      }

      requestAnimationFrame(render);
    }

    // ================= POINTER & TOUCH INTERACTIONS =================
    function handlePointer(e) {
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;

      mouse.targetX = clientX;
      mouse.targetY = clientY;

      addStardust(clientX, clientY, 1);

      if (mouse.isDown) {
        addStardust(clientX, clientY, 2);
      }
    }

    window.addEventListener('mousemove', handlePointer, { passive: true });
    window.addEventListener('touchmove', handlePointer, { passive: true });

    // Click/Touch Burst of Jasmine & Stardust
    window.addEventListener('mousedown', (e) => {
      if (e.target.closest('#ambientPopover, #toggleControlsBtn')) return;
      mouse.isDown = true;
      burstAt(e.clientX, e.clientY, 8);
    });

    window.addEventListener('mouseup', () => { mouse.isDown = false; });

    function burstAt(x, y, amount = 10) {
      addStardust(x, y, 16);
      for (let i = 0; i < amount; i++) {
        const b = new JasmineElement(false);
        b.x = x + (Math.random() - 0.5) * 50;
        b.y = y + (Math.random() - 0.5) * 50;
        const angle = Math.random() * Math.PI * 2;
        const speed = 2.5 + Math.random() * 5.5;
        b.vx = Math.cos(angle) * speed;
        b.vy = Math.sin(angle) * speed;
        blossoms.push(b);
      }
      if (blossoms.length > config.blossomCount + 30) {
        blossoms.splice(0, amount);
      }
    }

    // ================= CONTROLS LOGIC =================
    const toggleBtn = document.getElementById('toggleControlsBtn');
    const popover = document.getElementById('ambientPopover');
    const closeBtn = document.getElementById('closeControlsBtn');

    const petalSpeedInput = document.getElementById('petalSpeed');
    const petalDensityInput = document.getElementById('petalDensity');
    const sparkleIntensityInput = document.getElementById('sparkleIntensity');
    const toggleSwirlBtn = document.getElementById('toggleSwirlBtn');
    const scatterBtn = document.getElementById('scatterPetalsBtn');
    const paletteButtons = document.querySelectorAll('.palette-btn');

    function toggleMenu(show) {
      if (show) {
        popover.classList.remove('hidden');
        requestAnimationFrame(() => {
          popover.classList.remove('scale-95', 'opacity-0');
          popover.classList.add('scale-100', 'opacity-100');
        });
      } else {
        popover.classList.remove('scale-100', 'opacity-100');
        popover.classList.add('scale-95', 'opacity-0');
        setTimeout(() => popover.classList.add('hidden'), 250);
      }
    }

    toggleBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const isClosed = popover.classList.contains('hidden');
      toggleMenu(isClosed);
    });

    closeBtn.addEventListener('click', () => toggleMenu(false));

    // Breeze Speed Slider
    petalSpeedInput.addEventListener('input', (e) => {
      config.speedFactor = parseFloat(e.target.value);
      document.getElementById('speedVal').textContent =
        config.speedFactor > 1.8 ? 'Brisk' : config.speedFactor > 0.8 ? 'Languid' : 'Serene';
    });

    // Blossom Density Slider
    petalDensityInput.addEventListener('input', (e) => {
      config.blossomCount = parseInt(e.target.value, 10);
      document.getElementById('densityVal').textContent = `${config.blossomCount > 60 ? 'Opulent' : config.blossomCount > 30 ? 'Sumptuous' : 'Delicate'} (${config.blossomCount})`;
      syncBlossomCount();
    });

    // Amber Glow & Embers
    sparkleIntensityInput.addEventListener('input', (e) => {
      config.shimmerFactor = parseFloat(e.target.value);
      document.getElementById('shimmerVal').textContent =
        config.shimmerFactor > 1.8 ? 'Molten Glow' : config.shimmerFactor > 0.9 ? 'Radiant' : 'Subtle';
    });

    // Scent Profile / Palette Switcher
    paletteButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        paletteButtons.forEach(b => {
          b.classList.remove('active', 'border-amberlux-400/80', 'bg-caramel-900/60', 'text-[#fff5e6]');
          b.classList.add('border-transparent', 'bg-black/30', 'text-neutral-300');
        });
        btn.classList.add('active', 'border-amberlux-400/80', 'bg-caramel-900/60', 'text-[#fff5e6]');
        btn.classList.remove('border-transparent', 'bg-black/30', 'text-neutral-300');

        config.activePalette = btn.getAttribute('data-palette');
        blossoms.forEach(b => b.updateColor());
        embers.forEach(em => em.reset());
      });
    });

    // Toggle Vortex / Breeze Force
    toggleSwirlBtn.addEventListener('click', () => {
      config.isVortexActive = !config.isVortexActive;
      toggleSwirlBtn.textContent = config.isVortexActive ? 'Scent Vortex' : 'Gentle Breeze';
      toggleSwirlBtn.classList.toggle('border-amberlux-400');
      toggleSwirlBtn.classList.toggle('bg-amberlux-500/30');
    });

    // Scatter Action Button
    scatterBtn.addEventListener('click', () => {
      blossoms.forEach(b => {
        b.vx = (Math.random() - 0.5) * 11;
        b.vy = -3 - Math.random() * 7;
      });
      addStardust(width / 2, height / 2, 30);
    });

    // Close when clicking outside
    document.addEventListener('click', (e) => {
      if (!popover.contains(e.target) && !toggleBtn.contains(e.target)) {
        toggleMenu(false);
      }
    });

    window.addEventListener('resize', resize);

    window.onload = function() {
      resize();
      render();
    };
  </script>
