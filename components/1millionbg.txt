    // ================= LAYER 1: STARSTRUCK CELESTIAL CANVAS =================
    let stars = [];
    class Star {
      constructor() {
        this.reset();
      }
      reset() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.size = 0.5 + Math.random() * 1.8;
        this.twinkleSpeed = 0.012 + Math.random() * 0.035;
        this.phase = Math.random() * Math.PI * 2;
        this.baseAlpha = 0.25 + Math.random() * 0.75;
        this.isCross = Math.random() > 0.88; // Starstruck sparkle diffraction spike
      }
      update() {
        this.phase += this.twinkleSpeed;
      }
      draw(c) {
        const pal = palettes[config.activePalette];
        const alpha = Math.max(0.08, (Math.sin(this.phase) * 0.45 + 0.55) * this.baseAlpha * config.radianceFactor);
        const color = pal.stars[Math.floor(this.phase * 2) % pal.stars.length];

        c.save();
        c.globalAlpha = Math.min(1.0, alpha);
        c.fillStyle = color;
        c.shadowColor = color;
        c.shadowBlur = this.isCross ? 8 : 4;

        c.beginPath();
        c.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        c.fill();

        // 4-point starstruck diffraction spikes for hero celestial stars
        if (this.isCross && alpha > 0.4) {
          c.strokeStyle = color;
          c.lineWidth = 0.7;
          const spikeLen = this.size * 5 * (Math.sin(this.phase) * 0.3 + 0.7);

          c.beginPath();
          c.moveTo(this.x - spikeLen, this.y);
          c.lineTo(this.x + spikeLen, this.y);
          c.moveTo(this.x, this.y - spikeLen);
          c.lineTo(this.x, this.y + spikeLen);
          c.stroke();
        }

        c.restore();
      }
    }

    // ================= LAYER 2: 3D GOLD BULLION SHARDS & FLAKES =================
    class GoldElement {
      constructor(isInitial = false) {
        this.reset(isInitial);
      }

      reset(isInitial = false) {
        // Types: 0 = 24K Gold Geometric Ingot Shard, 1 = Tumbling Gold Leaf Flake, 2 = Suspended Metallic Dust / Ember
        const rand = Math.random();
        this.type = rand > 0.5 ? 0 : (rand > 0.22 ? 1 : 2);

        this.x = isInitial ? Math.random() * width : Math.random() * width * 1.2 - width * 0.1;
        this.y = isInitial ? Math.random() * height : height + 30 + Math.random() * 60;

        this.depth = 0.5 + Math.random() * 0.9;
        this.angle = Math.random() * Math.PI * 2;
        this.angularSpeed = (Math.random() - 0.5) * 0.04;

        if (this.type === 0) {
          // Angular 24K Gold Ingot / Bullion Shard
          this.size = (7 + Math.random() * 12) * this.depth;
          this.vx = (Math.random() - 0.5) * 0.6;
          this.vy = -(0.8 + Math.random() * 1.5) * this.depth; // Upward buoyant luxury drift
          this.flip = Math.random() * Math.PI;
          this.flipSpeed = 0.015 + Math.random() * 0.035;
          this.life = 1.0;
          this.decay = 0.0018 + Math.random() * 0.003;
          this.aspect = 0.45 + Math.random() * 0.5;
        } else if (this.type === 1) {
          // Organic Tumbling Gold Leaf Flake
          this.size = (5 + Math.random() * 8) * this.depth;
          this.vx = (Math.random() - 0.5) * 0.9;
          this.vy = -(1.0 + Math.random() * 1.8) * this.depth;
          this.flip = Math.random() * Math.PI;
          this.flipSpeed = 0.02 + Math.random() * 0.045;
          this.life = 1.0;
          this.decay = 0.002 + Math.random() * 0.0035;
        } else {
          // Molten Gold Micro-Ember / Metallic Sparkle
          this.size = (1.5 + Math.random() * 2.8) * this.depth;
          this.vx = (Math.random() - 0.5) * 0.8;
          this.vy = -(1.2 + Math.random() * 2.4) * this.depth;
          this.pulse = Math.random() * Math.PI * 2;
          this.life = 1.0;
          this.decay = 0.003 + Math.random() * 0.006;
        }

        this.updateColor();
      }

      updateColor() {
        const pal = palettes[config.activePalette];
        if (this.type === 0) {
          this.color = pal.shards[Math.floor(Math.random() * pal.shards.length)];
          this.edgeColor = '#ffffff';
        } else if (this.type === 1) {
          this.color = pal.shards[Math.floor(Math.random() * pal.shards.length)];
          this.edgeColor = '#fff9e6';
        } else {
          this.color = pal.dust[Math.floor(Math.random() * pal.dust.length)];
        }
      }

      update() {
        this.life -= this.decay;
        this.angle += this.angularSpeed * config.windFactor;

        this.x += this.vx * config.windFactor;
        this.y += this.vy * config.windFactor;

        if (this.type === 0 || this.type === 1) {
          this.flip += this.flipSpeed * config.windFactor;
        } else {
          this.pulse += 0.08;
          this.x += Math.sin(this.pulse) * 0.6;
        }

        // Magnetic Vortex Interaction
        const dx = this.x - mouse.x;
        const dy = this.y - mouse.y;
        const dist = Math.sqrt(dx * dx + dy * dy);

        if (dist < config.pullRadius) {
          const norm = 1 - dist / config.pullRadius;
          const angle = Math.atan2(dy, dx);

          if (config.isVortexActive) {
            // Swirl around pointer in golden orbit
            const tangent = angle + Math.PI / 2;
            const swirlStrength = norm * 4.8 * this.depth;
            const attraction = norm * 2.2 * this.depth;

            this.x += Math.cos(tangent) * swirlStrength - Math.cos(angle) * attraction;
            this.y += Math.sin(tangent) * swirlStrength - Math.sin(angle) * attraction;
          } else {
            // Direct repulsion
            const push = norm * 6.0 * this.depth;
            this.x += Math.cos(angle) * push;
            this.y += Math.sin(angle) * push;
          }
        }

        // Reset when leaving frame
        if (this.life <= 0 || this.y < -40 || this.x > width + 60 || this.x < -60) {
          this.reset(false);
        }
      }

      draw(c) {
        c.save();
        c.translate(this.x, this.y);
        c.rotate(this.angle);

        if (this.type === 0) {
          // 24K Gold Ingot / Shard with 3D bevel and specular metallic reflection
          const flipScale = Math.sin(this.flip);
          c.scale(1, Math.abs(flipScale) < 0.12 ? 0.12 : flipScale);
          c.globalAlpha = Math.max(0.1, this.life * 0.95);

          const w = this.size;
          const h = this.size * this.aspect;

          c.shadowColor = this.color;
          c.shadowBlur = 10 * config.radianceFactor;

          // Golden gradient body
          const grad = c.createLinearGradient(-w/2, -h/2, w/2, h/2);
          grad.addColorStop(0, '#fff5cc');
          grad.addColorStop(0.3, this.color);
          grad.addColorStop(0.7, '#b8860b');
          grad.addColorStop(1, '#664a04');

          c.fillStyle = grad;
          c.beginPath();
          c.moveTo(-w/2, -h/2);
          c.lineTo(w/2, -h/2 * 0.7);
          c.lineTo(w/2 * 0.8, h/2);
          c.lineTo(-w/2, h/2 * 0.8);
          c.closePath();
          c.fill();

          // Beveled metallic highlight rim
          c.strokeStyle = 'rgba(255, 255, 255, 0.65)';
          c.lineWidth = 0.8;
          c.stroke();

        } else if (this.type === 1) {
          // Gold Leaf Flake (Organic curved foil)
          const flipScale = Math.cos(this.flip);
          c.scale(Math.abs(flipScale) < 0.15 ? 0.15 : flipScale, 1);
          c.globalAlpha = Math.max(0.1, this.life * 0.88);

          c.shadowColor = this.color;
          c.shadowBlur = 8 * config.radianceFactor;

          c.fillStyle = this.color;
          c.beginPath();
          c.moveTo(0, -this.size * 0.7);
          c.bezierCurveTo(this.size * 0.6, -this.size * 0.4, this.size * 0.5, this.size * 0.6, 0, this.size * 0.7);
          c.bezierCurveTo(-this.size * 0.5, this.size * 0.4, -this.size * 0.6, -this.size * 0.3, 0, -this.size * 0.7);
          c.fill();

        } else {
          // Molten Gold Sparkle / Suspended Nectar Ember
          const emberAlpha = Math.max(0, this.life * (Math.sin(this.pulse) * 0.3 + 0.7));
          c.globalAlpha = emberAlpha;
          c.shadowColor = this.color;
          c.shadowBlur = 14 * config.radianceFactor;

          c.fillStyle = this.color;
          c.beginPath();
          c.arc(0, 0, this.size, 0, Math.PI * 2);
          c.fill();

          // Incandescent white core
          c.fillStyle = '#ffffff';
          c.beginPath();
          c.arc(0, 0, this.size * 0.38, 0, Math.PI * 2);
          c.fill();
        }

        c.restore();
      }
    }

    // ================= LAYER 3: INTERACTIVE FAST GOLD SPARK TRAIL =================
    const sparks = [];
    function addSparks(x, y, count = 1) {

          color: pal.sparks[Math.floor(Math.random() * pal.sparks.length)],
          life: 1.0,
          decay: 0.024 + Math.random() * 0.03
        });
      }
    }

    let elements = [];

    function syncShardCount() {
      if (elements.length < config.shardCount) {
        while (elements.length < config.shardCount) {
          elements.push(new GoldElement(false));
        }
      } else if (elements.length > config.shardCount) {
        elements.splice(config.shardCount);
      }
    }

    function resize() {
      width = starCanvas.width = mainCanvas.width = sparkCanvas.width = window.innerWidth;
      height = starCanvas.height = mainCanvas.height = sparkCanvas.height = window.innerHeight;

      stars = Array.from({ length: config.starCount }, () => new Star());
      elements = Array.from({ length: config.shardCount }, () => new GoldElement(true));
    }

    // ================= MAIN ANIMATION LOOP =================
    function render() {
      mouse.x += (mouse.targetX - mouse.x) * 0.08;
      mouse.y += (mouse.targetY - mouse.y) * 0.08;

      // Update Cursor Molten Gold Ambient Halo
      if (ambientGlow) {
        const xPercent = (mouse.x / width) * 100;
        const yPercent = (mouse.y / height) * 100;
        ambientGlow.style.setProperty('--mouse-x', `${xPercent}%`);
        ambientGlow.style.setProperty('--mouse-y', `${yPercent}%`);
      }

      // 1. Draw Starstruck Field
      starCtx.clearRect(0, 0, width, height);
      for (let s of stars) {
        s.update();
        s.draw(starCtx);
      }

      // 2. Draw 3D Floating Gold Shards & Flakes
      mainCtx.clearRect(0, 0, width, height);
      for (let i = 0; i < elements.length; i++) {
        elements[i].update();
        elements[i].draw(mainCtx);
      }

      // 3. Draw Stardust Trail Sparks
      sparkCtx.clearRect(0, 0, width, height);
      for (let i = sparks.length - 1; i >= 0; i--) {
        const sp = sparks[i];
        sp.x += sp.vx;
        sp.y += sp.vy;
        sp.life -= sp.decay;

        if (sp.life <= 0) {
          sparks.splice(i, 1);
          continue;
        }

        sparkCtx.save();
        sparkCtx.globalAlpha = sp.life * 0.95;
        sparkCtx.fillStyle = sp.color;
        sparkCtx.shadowColor = sp.color;
        sparkCtx.shadowBlur = 12;
        sparkCtx.beginPath();
        sparkCtx.arc(sp.x, sp.y, sp.radius * sp.life, 0, Math.PI * 2);
        sparkCtx.fill();
        sparkCtx.restore();
      }

      requestAnimationFrame(render);
    }

    // ================= POINTER & EXPLOSION INTERACTIONS =================
    function handlePointer(e) {
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;

      mouse.targetX = clientX;
      mouse.targetY = clientY;

      addSparks(clientX, clientY, 1);
      if (mouse.isDown) addSparks(clientX, clientY, 2);
    }

    window.addEventListener('mousemove', handlePointer, { passive: true });
    window.addEventListener('touchmove', handlePointer, { passive: true });

    window.addEventListener('mousedown', (e) => {
      if (e.target.closest('#ambientPopover, #toggleControlsBtn')) return;
      mouse.isDown = true;
      triggerSupernova(e.clientX, e.clientY, 18);
    });

    window.addEventListener('mouseup', () => { mouse.isDown = false; });

    // Supernova Burst Effect
    function triggerSupernova(x, y, count = 22) {
      addSparks(x, y, 30);
      for (let i = 0; i < count; i++) {
        const el = new GoldElement(false);
        el.x = x + (Math.random() - 0.5) * 35;
        el.y = y + (Math.random() - 0.5) * 35;
        const angle = Math.random() * Math.PI * 2;
        const speed = 3.5 + Math.random() * 7.5;
        el.vx = Math.cos(angle) * speed;
        el.vy = Math.sin(angle) * speed - 1.2;
        el.life = 1.0;
        el.size = (el.type === 2 ? 2.5 : 9) + Math.random() * 4;
        el.updateColor();
        elements.push(el);
      }

      if (elements.length > config.shardCount + 40) {
        elements.splice(0, count);
      }
    }

    // ================= UI CONSOLE LOGIC =================
    const toggleBtn = document.getElementById('toggleControlsBtn');
    const popover = document.getElementById('ambientPopover');
    const closeBtn = document.getElementById('closeControlsBtn');

    const windSpeedInput = document.getElementById('windSpeed');
    const shardDensityInput = document.getElementById('shardDensity');
    const starRadianceInput = document.getElementById('starRadiance');
    const toggleVortexBtn = document.getElementById('toggleVortexBtn');
    const triggerSupernovaBtn = document.getElementById('triggerSupernovaBtn');
    const paletteButtons = document.querySelectorAll('.palette-btn');

    function toggleMenu(show) {
      if (show) {
        popover.classList.remove('hidden');
        requestAnimationFrame(() => {
          popover.classList.remove('scale-95', 'opacity-0');
          popover.classList.add('scale-100', 'opacity-100');
        });
      } else {
        popover.classList.remove('scale-100', 'opacity-100');
        popover.classList.add('scale-95', 'opacity-0');
        setTimeout(() => popover.classList.add('hidden'), 250);
      }
    }

    toggleBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      const isClosed = popover.classList.contains('hidden');
      toggleMenu(isClosed);
    });

    closeBtn.addEventListener('click', () => toggleMenu(false));

    // Sliders
    windSpeedInput.addEventListener('input', (e) => {
      config.windFactor = parseFloat(e.target.value);
      document.getElementById('windVal').textContent = `${config.windFactor.toFixed(1)}x Flow`;
    });

    shardDensityInput.addEventListener('input', (e) => {
      config.shardCount = parseInt(e.target.value, 10);
      document.getElementById('densityVal').textContent = `${config.shardCount > 135 ? 'Max Bullion' : config.shardCount > 80 ? 'Rich' : 'Minimal'} (${config.shardCount})`;
      syncShardCount();
    });

    starRadianceInput.addEventListener('input', (e) => {
      config.radianceFactor = parseFloat(e.target.value);
      document.getElementById('radianceVal').textContent =
        config.radianceFactor > 2.0 ? 'Supernova' : config.radianceFactor > 1.1 ? 'Luminous' : 'Subtle Starfield';
    });

    // Palette Switcher
    paletteButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        paletteButtons.forEach(b => {
          b.classList.remove('active', 'border-amber-400/80', 'bg-amber-950/30', 'text-amber-200', 'shadow-[0_0_12px_rgba(255,215,0,0.15)]');
          b.classList.add('border-carbon/70', 'bg-onyx/60', 'text-slate-400');
        });
        btn.classList.add('active', 'border-amber-400/80', 'bg-amber-950/30', 'text-amber-200', 'shadow-[0_0_12px_rgba(255,215,0,0.15)]');
        btn.classList.remove('border-carbon/70', 'bg-onyx/60', 'text-slate-400');

        config.activePalette = btn.getAttribute('data-palette');
        elements.forEach(el => el.updateColor());
      });
    });

    // Dynamics Mode Toggle
    toggleVortexBtn.addEventListener('click', () => {
      config.isVortexActive = !config.isVortexActive;
      toggleVortexBtn.textContent = config.isVortexActive ? 'Magnetic Orbit' : 'Free Repulsion';
      toggleVortexBtn.classList.toggle('border-amber-400/40');
      toggleVortexBtn.classList.toggle('border-yellow-200/40');
    });

    // Supernova Trigger Button
    triggerSupernovaBtn.addEventListener('click', () => {
      triggerSupernova(width * 0.5, height * 0.5, 36);
      elements.forEach(el => {
        el.vy = -3.0 - Math.random() * 6.0;
        el.vx = (Math.random() - 0.5) * 8.0;
      });
    });

    // Close on outside click
    document.addEventListener('click', (e) => {
      if (!popover.contains(e.target) && !toggleBtn.contains(e.target)) {
        toggleMenu(false);
      }
    });

    window.addEventListener('resize', resize);

    window.onload = function() {
      resize();
      render();
    };
  </script>
